def fib(N)
{
  if(N == 0 | N == 1) {
    return 1;
  }
  else {
    return fib(N - 1) + fib (N - 2);
  }
}
def isPrime(N)
{
  factor = 2;
  if (N < 2) {
    return true;
  }
  while(factor <= N / 2)
  {
    if (N % factor == 0) {
      return false;
    }
    else {
      factor = factor + 1;
    }
  }
  return true;
}
print isPrime(fib(7));
